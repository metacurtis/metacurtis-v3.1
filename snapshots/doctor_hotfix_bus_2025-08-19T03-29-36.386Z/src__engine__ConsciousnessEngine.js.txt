// ConsciousnessEngine (SST v3.0) — builds blueprints on STAGE/QUALITY and emits BLUEPRINT_READY

import { Canonical } from '@config/canonical/canonicalAuthority.js';
import { createSeededRandom } from '../utils/random.js';

import { EVENTS } from '@theater/events.js';
import BeatBus from '@/modules/orchestration/core/BeatBusAdapter.js';

const BeatBus = (typeof getBeatBus === 'function' ? getBeatBus() : getBeatBus);

class ConsciousnessEngine {
  constructor() {
    this.blueprintCache = new Map();
    this.currentStage = 'genesis';
    this.currentQuality = 'HIGH';
    this.isInitialized = false;
    this.initializeBeatBusListeners();
    console.log('[Engine] ConsciousnessEngine ready (SST v3.0)');
  }

  initializeBeatBusListeners() {
    // Stage changes
    BeatBus.on(EVENTS.STAGE_CHANGE, (payload = {}) => { const stage = payload?.stage ?? payload?.to; if (!stage) return; 
      const stage = payload.stage ?? payload.to;
      if (!stage) return;
      console.log('[Engine] Stage change ->', stage);
      this.currentStage = stage;
      this.buildAndEmitBlueprint(stage, this.currentQuality);
    });

    // Quality changes
    BeatBus.on(EVENTS.QUALITY_CHANGE, (payload = {}) => { const tier = payload?.tier ?? payload?.quality; if (!tier) return; 
      const tier = payload.tier ?? payload.quality;
      if (!tier) return;
      console.log('[Engine] Quality change ->', tier);
      this.currentQuality = tier;
      this.buildAndEmitBlueprint(this.currentStage, tier);
    });

    // Prewarm genesis for opening
    BeatBus.on(EVENTS.PREWARM_GENESIS_BLUEPRINT, () => {
      const key = this._emergenceKey('HELLO CURTIS', 2000);
      if (!this.blueprintCache.has(key)) {
        const bp = this.buildEmergenceBlueprint({ text: 'HELLO CURTIS', count: 2000 });
        this.blueprintCache.set(key, bp);
      }
      BeatBus.emit(EVENTS.PREWARM_COMPLETE, { key });
    });

    // Build emergence blueprint on demand
    BeatBus.on(EVENTS.BUILD_EMERGENCE_BLUEPRINT, (opts = {}) => {
      const text = opts.sourceText || 'HELLO CURTIS';
      const count = opts.count || 2000;
      const key = this._emergenceKey(text, count);
      let bp = this.blueprintCache.get(key);
      const cached = !!bp;
      if (!bp) {
        bp = this.buildEmergenceBlueprint({ text, count });
        this.blueprintCache.set(key, bp);
      }
      BeatBus.emit(EVENTS.BLUEPRINT_READY, {
        blueprint: bp,
        stage: 'genesis',
        quality: this.currentQuality,
        cached,
      });
    });

    // Kick initial build after renderer mounts
    setTimeout(() => {
      console.log('[Engine] Initial state request');
      this.buildAndEmitBlueprint(this.currentStage, this.currentQuality);
    }, 500);
  }

  _emergenceKey(text, count) { return `emergence|${text}|${count}`; }

  buildAndEmitBlueprint(stage, quality) {
    const cacheKey = `${stage}|${quality}`;
    if (this.blueprintCache.has(cacheKey)) {
      const cachedBlueprint = this.blueprintCache.get(cacheKey);
      console.log('[Engine] Using cached blueprint', cacheKey);
      BeatBus.emit(EVENTS.BLUEPRINT_READY, { blueprint: cachedBlueprint, stage, quality, cached: true });
      return;
    }
    console.log('[Engine] Building blueprint', cacheKey);
    const blueprint = this.buildBlueprint(stage, { quality });
    if (blueprint) {
      this.blueprintCache.set(cacheKey, blueprint);
      BeatBus.emit(EVENTS.BLUEPRINT_READY, { blueprint, stage, quality, cached: false });
    }
  }

  buildBlueprint(stageName, options = {}) {
    const startTime = (typeof performance !== 'undefined' ? performance.now() : Date.now());
    const stageConfig = (Canonical && Canonical.stages && Canonical.stages[stageName]) || {};

    const SPEC_COUNTS = {
      genesis: 2000,
      discipline: 3000,
      neural: 5000,
      velocity: 12000,
      architecture: 8000,
      harmony: 12000,
      transcendence: 15000,
    };

    const quality = options.quality || this.currentQuality;
    const baseCount = Number.isFinite(stageConfig.particleCount) ? stageConfig.particleCount : (SPEC_COUNTS[stageName] || 5000);
    const particleCount = this.getParticleCountForQuality(baseCount, quality);
    const maxParticles = 15000;

    console.log(`[Engine] build ${stageName} count=${particleCount} quality=${quality}`);

    // Allocate
    const atmosphericPositions = new Float32Array(maxParticles * 3);
    const allenAtlasPositions   = new Float32Array(maxParticles * 3);
    const animationSeeds        = new Float32Array(maxParticles * 3);
    const sizeMultipliers       = new Float32Array(maxParticles);
    const opacityData           = new Float32Array(maxParticles);
    const atlasIndices          = new Float32Array(maxParticles);
    const tierData              = new Float32Array(maxParticles);

    // Tier config default
    const tierConfig = stageConfig.tierDistribution || {
      tier1: { ratio: 0.5, sizeRange: [0.5, 0.7], opacityRange: [0.3, 0.7] },
      tier2: { ratio: 0.2, sizeRange: [0.7, 0.9], opacityRange: [0.5, 0.8] },
      tier3: { ratio: 0.15, sizeRange: [1.0, 1.3], opacityRange: [0.7, 0.9] },
      tier4: { ratio: 0.15, sizeRange: [1.3, 2.0], opacityRange: [0.8, 1.0] },
    };

    const t1 = Math.floor(particleCount * tierConfig.tier1.ratio);
    const t2 = Math.floor(particleCount * tierConfig.tier2.ratio);
    const t3 = Math.floor(particleCount * tierConfig.tier3.ratio);
    const t4 = particleCount - t1 - t2 - t3;

    let idx = 0;
    this._generateTierParticles(atmosphericPositions, allenAtlasPositions, animationSeeds, sizeMultipliers, opacityData, atlasIndices, tierData, idx, t1, 0, tierConfig.tier1, stageName); idx += t1;
    this._generateTierParticles(atmosphericPositions, allenAtlasPositions, animationSeeds, sizeMultipliers, opacityData, atlasIndices, tierData, idx, t2, 1, tierConfig.tier2, stageName); idx += t2;
    this._generateTierParticles(atmosphericPositions, allenAtlasPositions, animationSeeds, sizeMultipliers, opacityData, atlasIndices, tierData, idx, t3, 2, tierConfig.tier3, stageName); idx += t3;
    this._generateTierParticles(atmosphericPositions, allenAtlasPositions, animationSeeds, sizeMultipliers, opacityData, atlasIndices, tierData, idx, t4, 3, tierConfig.tier4, stageName);

    const buildTime = (typeof performance !== 'undefined' ? performance.now() : Date.now()) - startTime;
    console.log(`[Engine] built in ${buildTime.toFixed(2)}ms`);

    return {
      stageName,
      particleCount,
      maxParticles,
      activeCount: particleCount,
      atmosphericPositions,
      allenAtlasPositions,
      animationSeeds,
      sizeMultipliers,
      opacityData,
      atlasIndices,
      tierData,
      metadata: { quality, buildTime, tierCounts: { tier1: t1, tier2: t2, tier3: t3, tier4: t4 } },
    };
  }

  buildEmergenceBlueprint({ text = 'HELLO CURTIS', count = 2000 } = {}) {
    const positions = this.textToParticlePositions(text, count);
    const tiers = new Uint8Array(count);
    const tierCounts = [ Math.floor(count * 0.6), Math.floor(count * 0.2), Math.floor(count * 0.1), Math.floor(count * 0.1) ];
    const total = tierCounts[0] + tierCounts[1] + tierCounts[2] + tierCounts[3];
    if (total < count) tierCounts[0] += (count - total);

    let k = 0;
    for (let t = 0; t < 4; t++) for (let i = 0; i < tierCounts[t] && k < count; i++) tiers[k++] = t;
    for (let i = count - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); const tmp = tiers[i]; tiers[i] = tiers[j]; tiers[j] = tmp; }

    return {
      id: 'emergence-genesis',
      mode: 'emergence',
      stageName: 'genesis',
      count,
      particleCount: count,
      maxParticles: count,
      activeCount: count,
      positions,
      tiers,
      metadata: { sourceText: text, createdAt: Date.now() },
    };
  }

  textToParticlePositions(text, count) {
    const positions = new Float32Array(count * 3);
    const charWidth = 10.0;
    const textHeight = 12.0;
    const centerY = 0;
    for (let i = 0; i < count; i++) {
      const charIndex = Math.floor(Math.random() * text.length);
      const baseX = (charIndex - text.length / 2) * charWidth;
      const x = baseX + (Math.random() - 0.5) * charWidth * 0.8;
      const y = centerY + (Math.random() - 0.5) * textHeight;
      const z = (Math.random() - 0.5) * 2.0;
      const j = i * 3; positions[j] = x; positions[j+1] = y; positions[j+2] = z;
    }
    return positions;
  }

  _generateTierParticles(atmos, allen, seeds, sizes, opac, atlas, tier, startIndex, count, tierIndex, tierCfg, stageName) {
    const rnd = createSeededRandom(`${stageName}-tier${tierIndex}`);
    const spriteByTier = { 0: [7], 1: [0,1], 2: [4], 3: [1] };
    const sprites = spriteByTier[tierIndex] || [0];

    for (let i = 0; i < count; i++) {
      const p3 = (startIndex + i) * 3;
      const p1 = startIndex + i;

      const r = rnd() * 80 + 40;
      const theta = rnd() * Math.PI * 2;
      const u = 2 * rnd() - 1;
      const phi = Math.acos(u);

      atmos[p3]   = r * Math.sin(phi) * Math.cos(theta);
      atmos[p3+1] = r * Math.sin(phi) * Math.sin(theta);
      atmos[p3+2] = r * Math.cos(phi);

      const br = 20 + rnd() * 15;
      allen[p3]   = br * Math.sin(phi) * Math.cos(theta);
      allen[p3+1] = br * Math.sin(phi) * Math.sin(theta);
      allen[p3+2] = br * Math.cos(phi);

      seeds[p3]   = rnd(); seeds[p3+1] = rnd(); seeds[p3+2] = rnd();

      const sizeMin = (tierCfg.sizeRange && tierCfg.sizeRange[0]) || 1.0;
      const sizeMax = (tierCfg.sizeRange && tierCfg.sizeRange[1]) || 1.0;
      const opMin   = (tierCfg.opacityRange && tierCfg.opacityRange[0]) || 0.8;
      const opMax   = (tierCfg.opacityRange && tierCfg.opacityRange[1]) || 1.0;

      sizes[p1] = sizeMin + rnd() * (sizeMax - sizeMin);
      opac[p1]  = opMin + rnd() * (opMax - opMin);
      atlas[p1] = sprites[Math.floor(rnd() * sprites.length)];
      tier[p1]  = tierIndex;
    }
  }

  getParticleCountForQuality(baseCount, quality) {
    const mult = { LOW: 0.3, MEDIUM: 0.6, HIGH: 1.0, ULTRA: 1.5 };
    const c = Math.floor(baseCount * (mult[quality] || 1.0));
    return Math.min(c, 15000);
  }

  clearCache() { this.blueprintCache.clear(); console.log('[Engine] cache cleared'); }

  getCacheStats() {
    const keys = Array.from(this.blueprintCache.keys());
    return {
      size: this.blueprintCache.size,
      keys,
      stages: keys.filter(k => !k.startsWith('emergence')).map(k => k.split('|')[0]),
      emergenceKeys: keys.filter(k => k.startsWith('emergence')),
    };
  }

  forceRebuild(stage, quality) {
    const key = `${stage}|${quality}`;
    this.blueprintCache.delete(key);
    console.log('[Engine] force rebuild', key);
    this.buildAndEmitBlueprint(stage, quality);
  }
}

const engine = new ConsciousnessEngine();

if (typeof window !== 'undefined') {
  window.engineDebug = {
    getCacheStats: () => engine.getCacheStats(),
    clearCache: () => engine.clearCache(),
    preGenerate: () => {
      const stages = Object.keys(Canonical.stages || {});
      const qualities = ['LOW','MEDIUM','HIGH','ULTRA'];
      stages.forEach(st => qualities.forEach(q => engine.buildAndEmitBlueprint(st, q)));
      console.log('[Engine] pre-generated all stage/quality combos');
    },
    forceRebuild: (stage, quality) => engine.forceRebuild(stage, quality),
  };
}

export default engine;
